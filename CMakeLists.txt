cmake_minimum_required(VERSION 3.30)
project(LogiOptionsOpen VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include(CTest)
include(GNUInstallDirs)
option(LOGI_BUILD_GUI "Build the native settings interface" ON)

if(WIN32)
    enable_language(RC)
    configure_file(default-windows.ini mappings.ini COPYONLY)
    add_library(logi-core STATIC hidpp.cpp mouse.cpp remap.cpp ring.cpp)
    target_link_libraries(logi-core PUBLIC setupapi hid shell32)
    add_executable(logi-cli main.cpp)
    target_link_libraries(logi-cli PRIVATE logi-core)
    set(logi_targets logi-core logi-cli)
    if(BUILD_TESTING)
        # Explicit opt-in hardware fault injection, never registered with CTest
        # or installed in application packages.
        add_executable(logi-windows-reporting-recovery-test tests/windows_reporting_recovery.cpp)
        target_include_directories(logi-windows-reporting-recovery-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(logi-windows-reporting-recovery-test PRIVATE logi-core)
        list(APPEND logi_targets logi-windows-reporting-recovery-test)
    endif()
    if(LOGI_BUILD_GUI)
        foreach(asset side.png front.png back.png loginative.ico firefox.png chrome.png wps.png)
            configure_file("assets/${asset}" "assets/${asset}" COPYONLY)
        endforeach()
        add_executable(logi-native WIN32 native.cpp smart_native.cpp assets.rc)
        target_link_libraries(logi-native PRIVATE logi-core comctl32 comdlg32 gdiplus ole32 dwmapi shlwapi uxtheme)
        list(APPEND logi_targets logi-native)
        install(TARGETS logi-native RUNTIME DESTINATION .)
    endif()
    foreach(target IN LISTS logi_targets)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
        if(MINGW)
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
            target_link_options(${target} PRIVATE -municode -static $<$<CONFIG:Release>:-s>)
        endif()
    endforeach()
    install(TARGETS logi-cli RUNTIME DESTINATION .)
    install(FILES "${CMAKE_BINARY_DIR}/mappings.ini" bridge.py DESTINATION .)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(linux)
    if(LOGI_BUILD_GUI)
        find_package(Qt6 REQUIRED COMPONENTS Widgets)
        add_executable(logi-options-open linux/gui.cpp linux/ring_ui.cpp)
        target_link_libraries(logi-options-open PRIVATE Qt6::Widgets)
        install(TARGETS logi-options-open RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
        if(BUILD_TESTING)
            find_package(Qt6 REQUIRED COMPONENTS Test)
            add_executable(logi-linux-gui-tests linux/gui.cpp linux/ring_ui.cpp)
            target_compile_definitions(logi-linux-gui-tests PRIVATE LOGI_GUI_TEST)
            target_link_libraries(logi-linux-gui-tests PRIVATE Qt6::Widgets Qt6::Test)
            add_test(NAME linux-gui-interaction COMMAND logi-linux-gui-tests)
            set_tests_properties(linux-gui-interaction PROPERTIES TIMEOUT 90)
            add_test(NAME linux_gui_offscreen COMMAND logi-options-open --self-test)
            set_tests_properties(linux_gui_offscreen PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        endif()
    endif()
    install(PROGRAMS bridge.py DESTINATION ${CMAKE_INSTALL_DATADIR}/logi-options-open)
else()
    message(FATAL_ERROR "Supported platforms: Windows and Linux")
endif()
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME bridge_consent_and_platform COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_bridge.py")
    endif()
    add_executable(first-run-test tests/first_run.cpp)
    if(WIN32)
        target_link_libraries(first-run-test PRIVATE shell32)
        if(MINGW)
            target_link_options(first-run-test PRIVATE -static)
        endif()
    endif()
    add_test(NAME first_run_notice COMMAND first-run-test)
endif()
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_PACKAGE_NAME "logi-options-open")
set(CPACK_PACKAGE_VERSION "0.1.0-alpha.2")
set(CPACK_PACKAGE_VENDOR "Logi Options Open contributors")
set(CPACK_GENERATOR "ZIP")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
