cmake_minimum_required(VERSION 3.30)
project(LogiOptionsOpen VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include(CTest)
include(GNUInstallDirs)
if(WIN32)
    set(CMAKE_INSTALL_BINDIR ".")
endif()
option(LOGI_BUILD_GUI "Build the native settings interface" ON)

if(WIN32)
    enable_language(RC)
    configure_file(default-windows.ini mappings.ini COPYONLY)
    add_library(logi-core STATIC hidpp.cpp mouse.cpp remap.cpp ring.cpp)
    target_link_libraries(logi-core PUBLIC setupapi hid shell32)
    add_executable(logi-cli main.cpp)
    target_link_libraries(logi-cli PRIVATE logi-core)
    set(logi_targets logi-core logi-cli)
    if(BUILD_TESTING)
        # Explicit opt-in hardware fault injection, never registered with CTest
        # or installed in application packages.
        add_executable(logi-windows-reporting-recovery-test tests/windows_reporting_recovery.cpp)
        target_include_directories(logi-windows-reporting-recovery-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(logi-windows-reporting-recovery-test PRIVATE logi-core)
        list(APPEND logi_targets logi-windows-reporting-recovery-test)
    endif()
    set(logi_gui_target logi-native)
    foreach(target IN LISTS logi_targets)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
        if(MINGW)
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
            target_link_options(${target} PRIVATE -municode -static $<$<CONFIG:Release>:-s>)
        endif()
    endforeach()
    install(TARGETS logi-cli RUNTIME DESTINATION .)
    install(FILES "${CMAKE_BINARY_DIR}/mappings.ini" bridge.py DESTINATION .)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(linux)
    set(logi_gui_target logi-options-open)
    install(PROGRAMS bridge.py DESTINATION ${CMAKE_INSTALL_DATADIR}/logi-options-open)
else()
    message(FATAL_ERROR "Supported platforms: Windows and Linux")
endif()
if(LOGI_BUILD_GUI)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    qt_add_resources(LOGI_UI_RESOURCES ui-assets.qrc)
    set(LOGI_UI_SOURCES ui/app.cpp ui/platform.cpp ui/ring_ui.cpp ui/device_canvas.cpp ${LOGI_UI_RESOURCES})
    qt_add_executable(${logi_gui_target} ${LOGI_UI_SOURCES})
    target_link_libraries(${logi_gui_target} PRIVATE Qt6::Widgets)
    if(WIN32)
        set_target_properties(${logi_gui_target} PROPERTIES WIN32_EXECUTABLE TRUE)
        target_compile_definitions(${logi_gui_target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(${logi_gui_target} PRIVATE shell32)
        target_sources(${logi_gui_target} PRIVATE assets.rc)
    endif()
    install(TARGETS ${logi_gui_target} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    if(WIN32)
        qt_generate_deploy_app_script(TARGET ${logi_gui_target} OUTPUT_SCRIPT LOGI_QT_DEPLOY NO_TRANSLATIONS)
        install(SCRIPT ${LOGI_QT_DEPLOY})
    endif()
    if(BUILD_TESTING)
        find_package(Qt6 REQUIRED COMPONENTS Test)
        if(WIN32)
            set(logi_gui_test logi-windows-gui-tests)
        else()
            set(logi_gui_test logi-linux-gui-tests)
        endif()
        add_executable(${logi_gui_test} ${LOGI_UI_SOURCES})
        target_compile_definitions(${logi_gui_test} PRIVATE LOGI_GUI_TEST)
        target_link_libraries(${logi_gui_test} PRIVATE Qt6::Widgets Qt6::Test)
        if(WIN32)
            target_compile_definitions(${logi_gui_test} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
            target_link_libraries(${logi_gui_test} PRIVATE shell32)
        endif()
        add_test(NAME shared-gui-interaction COMMAND ${logi_gui_test})
        set_tests_properties(shared-gui-interaction PROPERTIES TIMEOUT 90)
        add_test(NAME shared-ring-standalone COMMAND ${logi_gui_test} --standalone-ring)
        set_tests_properties(shared-ring-standalone PROPERTIES TIMEOUT 30)
        add_test(NAME shared-gui-offscreen COMMAND ${logi_gui_target} --self-test)
        set_tests_properties(shared-gui-offscreen PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        add_executable(logi-ui-platform-tests ui/platform.cpp ui/platform_test.cpp)
        target_link_libraries(logi-ui-platform-tests PRIVATE Qt6::Core)
        add_test(NAME shared-ui-platform COMMAND logi-ui-platform-tests)
    endif()
endif()
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME bridge_consent_and_platform COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_bridge.py")
    endif()
    add_executable(first-run-test tests/first_run.cpp)
    if(WIN32)
        target_link_libraries(first-run-test PRIVATE shell32)
        if(MINGW)
            target_link_options(first-run-test PRIVATE -static)
        endif()
    endif()
    add_test(NAME first_run_notice COMMAND first-run-test)
endif()
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_PACKAGE_NAME "logi-options-open")
set(CPACK_PACKAGE_VERSION "0.1.0-alpha.2")
set(CPACK_PACKAGE_VENDOR "Logi Options Open contributors")
set(CPACK_GENERATOR "ZIP")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
