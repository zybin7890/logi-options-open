find_package(PkgConfig REQUIRED)
pkg_check_modules(HIDAPI REQUIRED IMPORTED_TARGET hidapi-hidraw)
find_package(Threads REQUIRED)
find_package(X11 REQUIRED COMPONENTS Xtst)
add_library(logi-linux-core STATIC hidpp_linux.cpp ../mouse.cpp agent.cpp desktop.cpp)
target_include_directories(logi-linux-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(logi-linux-core PUBLIC PkgConfig::HIDAPI Threads::Threads X11::X11 X11::Xtst)
target_compile_options(logi-linux-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(logi-cli cli.cpp)
target_link_libraries(logi-cli PRIVATE logi-linux-core)
set_target_properties(logi-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
configure_file(default-linux.ini "${CMAKE_BINARY_DIR}/default-linux.ini" COPYONLY)
install(TARGETS logi-cli RUNTIME DESTINATION bin)
install(FILES default-linux.ini DESTINATION share/logi-options-open)
