function(logi_wayland_protocol target protocol)
  set(xml "${CMAKE_CURRENT_SOURCE_DIR}/kde/protocols/${protocol}.xml")
  set(header "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-client-protocol.h")
  set(source "${CMAKE_CURRENT_BINARY_DIR}/${protocol}-protocol.c")
  add_custom_command(OUTPUT "${header}" COMMAND "${WAYLAND_SCANNER}" client-header "${xml}" "${header}" DEPENDS "${xml}")
  add_custom_command(OUTPUT "${source}" COMMAND "${WAYLAND_SCANNER}" private-code "${xml}" "${source}" DEPENDS "${xml}")
  add_library(${target} STATIC "${source}" "${header}")
  target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(${target} PUBLIC PkgConfig::WAYLAND_CLIENT)
endfunction()
