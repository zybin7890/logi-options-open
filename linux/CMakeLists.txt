find_package(PkgConfig REQUIRED)
pkg_check_modules(HIDAPI REQUIRED IMPORTED_TARGET hidapi-hidraw)
find_package(Threads REQUIRED)
add_library(logi-linux-core STATIC hidpp_linux.cpp ../mouse.cpp agent.cpp)
target_include_directories(logi-linux-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(logi-linux-core PUBLIC PkgConfig::HIDAPI Threads::Threads)
target_compile_options(logi-linux-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(logi-cli cli.cpp)
target_link_libraries(logi-cli PRIVATE logi-linux-core)
set_target_properties(logi-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
configure_file(default-linux.ini "${CMAKE_BINARY_DIR}/default-linux.ini" COPYONLY)
install(TARGETS logi-cli RUNTIME DESTINATION bin)
install(FILES default-linux.ini DESTINATION share/logi-options-open)
install(FILES 70-logi-options-open.rules logi-options-open.service DESTINATION share/logi-options-open/linux)
if(BUILD_TESTING)
  add_executable(logi-linux-tests tests.cpp)
  target_link_libraries(logi-linux-tests PRIVATE logi-linux-core)
  add_test(NAME linux-mapping-protocol COMMAND logi-linux-tests)
  add_test(NAME linux-default-config COMMAND logi-cli check-config "${CMAKE_BINARY_DIR}/default-linux.ini")
  add_test(NAME linux-help COMMAND logi-cli --help)
endif()
